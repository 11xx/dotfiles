#+begin_src yaml-ts
- name: 'aur: Install autofs'
  aur:
    name: autofs
    use: paru
    state: present
  become_user: aurbuilder

- name: 'fs: Create gak NFS base mount point'
  ansible.builtin.file:
    path: /gak
    state: directory
    owner: root
    group: root
    mode: "0755"

- name: 'autofs: Register /gak in auto.master.d'
  ansible.builtin.copy:
    dest: /etc/autofs/auto.master.d/gak.autofs
    content: "/gak /etc/autofs/auto.gak --timeout=3600 --ghost"

    owner: root
    group: root
    mode: "0644"
  notify: Reload autofs

- name: 'autofs: Configure gak NFS mounts'
  ansible.builtin.copy:
    dest: /etc/autofs/auto.gak
    # soft     : return EIO after retrans failures instead of hanging forever
    # timeo=5  : 0.5s initial RPC timeout before retry (tenths of a second)
    # retrans=2: give up after 2 retries (~1.5s total before soft error)
    # retry=0  : do not retry the mount attempt if server is unreachable
    content: |
      music -fstype=nfs4,ro,soft,timeo=5,retrans=2,retry=0  gak:/turak/music
      turak -fstype=nfs4,rw,soft,timeo=5,retrans=2,retry=0  gak:/turak
      marak -fstype=nfs4,rw,soft,timeo=5,retrans=2,retry=0  gak:/marak
      media -fstype=nfs4,rw,soft,timeo=5,retrans=2,retry=0  gak:/media
    owner: root
    group: root
    mode: "0644"
  notify: Reload autofs

- name: 'autofs: Register local drives in auto.master.d'
  ansible.builtin.copy:
    dest: /etc/autofs/auto.master.d/local-drives.autofs
    content: "/- /etc/autofs/auto.local-drives --timeout=1800"
    owner: root
    group: root
    mode: "0644"
  notify: Reload autofs

- name: 'autofs: Configure local mounts'
  ansible.builtin.copy:
    dest: /etc/autofs/auto.local-drives
    content: |
      /games -fstype=auto,rw,noatime,compress=zstd:1,ssd,space_cache=v2,discard=async,commit=120 :LABEL=GAMES
    owner: root
    group: root
    mode: "0644"
  notify: Reload autofs

- name: 'systemd: Ensure autofs drop-in directory exists'
  ansible.builtin.file:
    path: /etc/systemd/system/autofs.service.d
    state: directory
    owner: root
    group: root
    mode: "0755"

- name: 'systemd: Enable and start autofs.service'
  ansible.builtin.systemd_service:
    name: autofs.service
    enabled: true
    state: started
#+end_src

- name: 'systemd: Fix autofs.service dependency race after boot'
  ansible.builtin.copy:
    dest: /etc/systemd/system/autofs.service.d/dependency-race-fix.conf
    content: |
      [Unit]
      Requires=network.target rpc-statd.service rpcbind.service
    owner: root
    group: root
    mode: "0644"
  notify: Reload systemd daemon

#+begin_src yaml-ts
- name: Install UFW
  ansible.builtin.package:
    name: ufw
    state: present
  become: true

- name: Reset UFW to defaults before applying rules
  community.general.ufw:
    state: reset
  become: true
  when: not ansible_check_mode

- name: Set default policies
  community.general.ufw:
    direction: "{{ item.direction }}"
    policy: "{{ item.policy }}"
  loop:
    - { direction: incoming, policy: deny }
    - { direction: outgoing, policy: allow }
  become: true

- name: Allow loopback interface
  community.general.ufw:
    rule: allow
    interface: lo
    direction: in
  become: true

- name: Apply host-specific UFW rules
  community.general.ufw:
    rule: "{{ item.rule | default('allow') }}"
    port: "{{ item.port | default(omit) }}"
    proto: "{{ item.proto | default(omit) }}"
    src: "{{ item.src | default(omit) }}"
    interface: "{{ item.interface | default(omit) }}"
    direction: "{{ item.direction | default('in') }}"
    comment: "{{ item.comment | default(omit) }}"
  loop: "{{ ufw_rules }}"
  become: true

- name: Enable UFW
  community.general.ufw:
    state: enabled
    logging: on
  become: true
#+end_src

      if test -d /etc/makepkg.conf.d/; then

** Package placement reference

Packages stay in the role that can independently need them. Duplicate
declarations are fine because package installation is idempotent.

| Package kind | Preferred Ansible home |
|--------------+------------------------|
| Boot, kernels, base system, pacman/admin tools | =misc_packages= unless a system file role configures them |
| Filesystems, disks, mounts, recovery utilities | =file_system= when configured there, otherwise =misc_packages= |
| Firewall, SSH, VPN, network services | =firewall=, =services=, =vpn=, or =misc_packages= for unconfigured tools |
| Fonts and font families | =fonts= or =misc_packages= for kept/experimental families |
| Shell, terminal, CLI/TUI utilities | =shell= when part of interactive defaults, otherwise =misc_packages= |
| Hyprland, Wayland, portal, screenshot, cursor helpers | =hyprland= when tied to the WM, otherwise =misc_packages= |
| Desktop apps, themes, file managers, media viewers | =desktop= or =browser= for browsers |
| Android tooling | =android= |
| MPD, music playback, tagging, scrobbling | =mpd= when tied to the music setup, otherwise =misc_packages= |
| Containers and NVIDIA container integration | =containers= |
| QEMU/KVM/libvirt/VM tooling | =qemu-kvm= or =controller-vm= when configured there |
| AI-agent host requirements | =aiagent_host_setup= |
| Explicit packages without a strong owner | =misc_packages= |

      # VirtIO:  driver ISO attached as second CD so it can be installed

        -drive if=pflash,format=raw,readonly=on,file={{ ovmf_code }} \
        -drive if=pflash,format=raw,file="${VM_DIR}/OVMF_VARS.fd" \

- name: Install VM start script
  ansible.builtin.copy:
    content: |
      #!/usr/bin/env bash
      # Managed by Ansible — do not edit by hand
      set -euo pipefail

      VM_DIR="{{ xbox_vm_dir }}"
      cd "$VM_DIR"

      usb_args=()

      add_usb_if_present() {
        local vendor="$1"
        local product="$2"
        local label="$3"

        if lsusb -d "${vendor}:${product}" >/dev/null 2>&1; then
          echo "Passing through ${label}: ${vendor}:${product}"
          usb_args+=(
            -device "usb-host,bus=xhci.0,vendorid=0x${vendor},productid=0x${product}"
          )
        else
          echo "Not present: ${label}: ${vendor}:${product}"
        fi
      }

      # Release host drivers that may bind Xbox-like controller paths.
      sudo modprobe -r xone 2>/dev/null || true
      sudo modprobe -r xpad 2>/dev/null || true

      # # Do not reload xpad automatically here; it may immediately grab 8BitDo again.
      # trap 'sudo modprobe xone 2>/dev/null || true' EXIT

      add_usb_if_present "{{ xbox_dongle_vendor }}" "{{ xbox_dongle_product }}" "Xbox Wireless Adapter"
      add_usb_if_present "{{ xbox_controller_vendor }}" "{{ xbox_controller_product }}" "Xbox wired controller"

      add_usb_if_present "{{ bitdo_vendor }}" "{{ bitdo_receiver_active_product }}" "8BitDo UM2 receiver active mode"
      add_usb_if_present "{{ bitdo_vendor }}" "{{ bitdo_receiver_idle_product }}" "8BitDo UM2 receiver idle/update mode"
      add_usb_if_present "{{ nintendo_vendor }}" "{{ nintendo_pro_product }}" "8BitDo/Switch Pro mode"
      add_usb_if_present "{{ vader5_vendor }}" "{{ vader5_product }}" "Flydigi Vader 5 Pro"

      exec qemu-system-x86_64 \
        -enable-kvm \
        -cpu host \
        -machine type=q35,accel=kvm \
        -m {{ xbox_vm_ram }} \
        -drive if=pflash,format=raw,readonly=on,file={{ ovmf_code }} \

        -display gtk,grab-on-hover=on \
        -device qemu-xhci,id=xhci \
        -device usb-tablet \
        "${usb_args[@]}" \
        -boot order=c
    dest: "{{ xbox_vm_dir }}/start-controller-vm.sh"
    mode: "0750"

# https://github.com/eza-community/eza/blob/main/INSTALL.md#debian-and-ubuntu
- name: Update apt cache
  ansible.builtin.apt:
    update_cache: yes
  become: yes

- name: Install gpg package
  ansible.builtin.apt:
    name: gpg
    state: present
  become: yes

- name: Create /etc/apt/keyrings directory
  ansible.builtin.file:
    path: /etc/apt/keyrings
    state: directory
    mode: '0755'
  become: yes

- name: Download and dearmor eza GPG key
  ansible.builtin.command:
    cmd: >
      bash -c "wget -qO- https://raw.githubusercontent.com/eza-community/eza/main/deb.asc
      | gpg --dearmor -o /etc/apt/keyrings/gierens.gpg"
    creates: /etc/apt/keyrings/gierens.gpg
  become: yes

- name: Add eza apt repository
  ansible.builtin.apt_repository:
    repo: "deb [signed-by=/etc/apt/keyrings/gierens.gpg] http://deb.gierens.de stable main"
    state: present
    filename: gierens
  become: yes

- name: Set permissions on eza key and repository files
  ansible.builtin.file:
    path: "{{ item }}"
    mode: '0644'
  loop:
    - /etc/apt/keyrings/gierens.gpg
    - /etc/apt/sources.list.d/gierens.list
  become: yes

- name: Update apt cache after adding repository
  ansible.builtin.apt:
    update_cache: yes
  become: yes

- name: Install eza package
  ansible.builtin.apt:
    name: eza
    state: present
  become: yes
