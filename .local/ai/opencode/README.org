#+NAME: searxng-port
#+begin_src emacs-lisp :exports none
(defun searxng-port ()
  "Return SEARXNG_PORT value from the .env file as a string."
  (with-temp-buffer
    (insert-file-contents "~/ai/searxng/.env")
    (when (re-search-forward "^SEARXNG_PORT=\\([0-9]+\\)" nil t)
      (match-string 1))))

(searxng-port)
#+end_src

#+begin_src shell
# interactive shell as aiagent
# ======================================================================
sudo machinectl shell aiagent@

# systemd user unit management
# ======================================================================
# reload after deploying new/changed Quadlet files
sudo machinectl shell aiagent@ /usr/bin/systemctl --user daemon-reload

# start / stop / restart / status
sudo machinectl shell aiagent@ /usr/bin/systemctl --user start    opencode.service
sudo machinectl shell aiagent@ /usr/bin/systemctl --user stop     opencode.service
sudo machinectl shell aiagent@ /usr/bin/systemctl --user restart  opencode.service
sudo machinectl shell aiagent@ /usr/bin/systemctl --user status   opencode.service

# same for searxng-mcp
sudo machinectl shell aiagent@ /usr/bin/systemctl --user restart  searxng-mcp.service

# Podman as aiagent
# ======================================================================
sudo machinectl shell --uid=aiagent .host -- /usr/bin/podman ps
sudo machinectl shell --uid=aiagent .host -- /usr/bin/podman images
sudo machinectl shell --uid=aiagent .host -- /usr/bin/podman logs opencode
sudo machinectl shell --uid=aiagent .host -- /usr/bin/podman system prune -a

# build the opencode image manually (if not using opencode.build Quadlet)
sudo machinectl shell --uid=aiagent .host -- \
     /usr/bin/podman build \
     -t localhost/opencode:latest \
     /home/aiagent/opencode/

# non-interactive scripting (Ansible handler pattern)
# ======================================================================
sudo systemd-run --quiet --pipe --wait \
     --user --machine=aiagent@.host \
     /usr/bin/systemctl --user daemon-reload
#+end_src
