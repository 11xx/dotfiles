FROM docker.io/node:22-bookworm-slim

ARG OPENCODE_NPM_PACKAGE="opencode-ai"

# Essentials for opencode:
# - Network + VCS
# - Text search and JSON tools
# - Editors
# - Minimal build toolchain
RUN apt-get update && apt-get install -y --no-install-recommends \
    # Network & VCS (required by opencode workflows)
    ca-certificates \
    curl \
    git \
    openssh-client \
    wget \
    # Core shell utilities
    less \
    procps \
    unzip \
    xz-utils \
    # JSON / search
    jq \
    ripgrep \
    # Editors
    emacs-nox \
    nano \
    vim \
    # Build toolchain (for native addons, small utilities, etc.)
    g++ \
    make \
    python3 \
    # Additional CLI tooling (recommended but not strictly required)
    # Code navigation
    fd-find \
    universal-ctags \
    tree \
    # Code reading
    bat \
    # Linting & validation
    shellcheck \
    strace \
    entr \
    # Data utilities
    sqlite3 \
    python3-csvkit \
  && rm -rf /var/lib/apt/lists/*

# Tools not in bookworm repos — install via npm or binary
RUN npm install -g @ast-grep/cli

RUN curl -sSL https://github.com/dandavison/delta/releases/latest/download/delta-$(curl -sSL https://api.github.com/repos/dandavison/delta/releases/latest | jq -r .tag_name)-x86_64-unknown-linux-musl.tar.gz \
    | tar -xz --wildcards --strip-components=1 -C /usr/local/bin '*/delta'

RUN curl -sSL https://github.com/hadolint/hadolint/releases/latest/download/hadolint-Linux-x86_64 \
    -o /usr/local/bin/hadolint && chmod +x /usr/local/bin/hadolint

RUN curl -sSL https://github.com/mikefarah/yq/releases/latest/download/yq_linux_amd64 \
    -o /usr/local/bin/yq && chmod +x /usr/local/bin/yq

RUN npm install -g @ast-grep/cli


USER node
WORKDIR /workspace

ENV NPM_CONFIG_PREFIX=/home/node/.npm-global
ENV PATH=/home/node/.npm-global/bin:$PATH
ENV HOME=/home/node

RUN mkdir -p /home/node/.npm-global \
    /home/node/.config/opencode \
    /home/node/.local/share/opencode \
    /home/node/.local/state/opencode \
    /workspace && \
    npm install -g "${OPENCODE_NPM_PACKAGE}" && \
    opencode --version

COPY --chown=node:node entrypoint.sh /home/node/entrypoint.sh
RUN chmod +x /home/node/entrypoint.sh

ENTRYPOINT ["/home/node/entrypoint.sh"]
CMD ["opencode", "/workspace"]
#+end_src

#+begin_src systemd
[Unit]
Description=Opencode AI Container

[Container]
Image=opencode.build
AutoUpdate=local
ContainerName=opencode
WorkingDir=/workspace
UserNS=keep-id
Environment=HOME=/home/node
Volume=<<opencode-home>>/workspace:/workspace:Z
Volume=<<opencode-home>>/config:/home/node/.config/opencode:Z
Volume=<<opencode-home>>/share:/home/node/.local/share/opencode:Z
Volume=<<opencode-home>>/state:/home/node/.local/state/opencode:Z
Exec=opencode /workspace

[Install]
WantedBy=default.target
#+end_src
