FROM quay.io/fedora/fedora:43

ARG OPENCODE_VERSION="latest"

RUN dnf -y upgrade --refresh

RUN dnf -y install --setopt=install_weak_deps=False \
    # === Essentials for OpenCode ===
    # Required for Node environment, native addon builds, VCS, and core agent tools
    ca-certificates \
    curl \
    findutils \
    gcc-c++ \
    git \
    gzip \
    jq \
    make \
    nodejs \
    npm \
    openssh-clients \
    procps-ng \
    python3 \
    ripgrep \
    shadow-utils \
    tar \
    unzip \
    wget \
    which \
    xz \
    # === Arbitrary Additional Tools ===
    # Optional editors, linters, and CLI utilities (safe to modify or remove)
    bat \
    emacs \
    entr \
    git-delta \
    less \
    nano \
    shellcheck \
    sqlite \
    strace \
    tree \
    uv \
    vim-enhanced \
    yq \
    ctags \
 && dnf clean all \
 && rm -rf /var/cache/dnf

# Global Node-based tools
RUN npm install -g @ast-grep/cli

# Hadolint (Haskell binary not in Fedora core)
RUN curl -sSL https://github.com/hadolint/hadolint/releases/latest/download/hadolint-Linux-x86_64 \
    -o /usr/local/bin/hadolint \
 && chmod +x /usr/local/bin/hadolint

RUN groupadd -g 1000 node \
 && useradd -m -u 1000 -g 1000 -s /bin/bash node

USER node
WORKDIR /workspace

ENV NPM_CONFIG_PREFIX=/home/node/.npm-global
ENV PATH=/home/node/.npm-global/bin:$PATH
ENV HOME=/home/node

RUN mkdir -pv \
    /home/node/.npm-global \
    /home/node/.config/opencode \
    /home/node/.local/share/opencode \
    /home/node/.local/state/opencode \
    /workspace

RUN npm install -g "opencode-ai@${OPENCODE_VERSION}" \
 && opencode --version \
 && emacs --version | head -n 1 \
 && node --version \
 && npm --version \
 && uv --version

COPY --chown=node:node entrypoint.sh /home/node/entrypoint.sh
RUN chmod +x /home/node/entrypoint.sh

ENTRYPOINT ["/home/node/entrypoint.sh"]
CMD ["opencode", "/workspace"]
#+end_src

#+begin_src systemd
[Unit]
Description=Opencode AI Container

[Container]
Image=opencode.build
AutoUpdate=local
ContainerName=opencode
WorkingDir=/workspace
UserNS=keep-id
Environment=HOME=/home/node
Volume=<<opencode-home>>/workspace:/workspace:Z
Volume=<<opencode-home>>/config:/home/node/.config/opencode:Z
Volume=<<opencode-home>>/share:/home/node/.local/share/opencode:Z
Volume=<<opencode-home>>/state:/home/node/.local/state/opencode:Z
Exec=opencode /workspace

[Install]
WantedBy=default.target
#+end_src
