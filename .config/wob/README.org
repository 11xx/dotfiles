** =wob-volume=: Script to setup wob for volume
:PROPERTIES:
:header-args:shell: :tangle (tdir "wob-volume") :shebang #!/usr/bin/env sh
:CUSTOM_ID: orgid-snwuth

This script monitors the config file from the heredoc and uses entr to
restart tail + wob.

- DEPENDENCIES:
  + wob
  + entr

#+begin_src shell
set -e
while :
do case $1 in
       -v)
           set -x
           ;;
       ,*) break
   esac; shift; done

wobPipeVolume=/tmp/wobpipe-volume
[ -p "${wobPipeVolume}" ] || mkfifo "${wobPipeVolume}"
# tail -f $wobPipeVolume | wob
killall wob || echo 'No wob instance found, skipping...'
entr -r -- sh -c "tail -f \"${wobPipeVolume}\" | wob" <<EOF
${XDG_CONFIG_HOME}/wob/wob.ini
EOF
#+end_src

** =wob-instance=: Instance monitor starter for custom pipes and config
:PROPERTIES:
:header-args:shell: :tangle (tdir "wob-instance") :shebang #!/usr/bin/env sh
:END:

#+begin_src shell
#!/usr/bin/env sh
# Usage: start-wob-instance <PIPE_PATH> <CONFIG_FILE>

PIPE="$1"
CONFIG="$2"

if [ -z "$PIPE" ] || [ -z "$CONFIG" ]; then
    echo "Usage: $0 PIPE_PATH CONFIG_FILE"
    exit 1
fi

# Ensure the named pipe exists
[ -p "$PIPE" ] || mkfifo "$PIPE"

# Clean up pipe on exit
trap "rm -f $PIPE" EXIT

# Start wob reading from the pipe
# tail -f keeps the pipe open even after a write finishes
tail -f "$PIPE" | wob -c "$CONFIG"
#+end_src
