** =wob-volume=: Script to setup wob for volume
:PROPERTIES:
:header-args:shell: :tangle (tdir "wob-volume") :shebang #!/usr/bin/env sh
:CUSTOM_ID: orgid-snwuth
:END:

Monitor =wob='s config file and restart on change with =entr=

- DEPENDENCIES:
  + wob
  + entr

#+begin_src shell
wobPipe="${XDG_RUNTIME_DIR:-/tmp}/wobpipe-volume"
wobConfig="${XDG_CONFIG_HOME:-$HOME/.config}/wob/wob.ini"
wobPidFile="/tmp/wob-volume.pid$$"
#+end_src

** =wob-instance=: Instance monitor starter for custom pipes and config
:PROPERTIES:
:header-args:shell: :tangle (tdir "wob-instance") :shebang #!/usr/bin/env sh
:END:

#+begin_src shell
wobPipe="$1"
wobConfig="$2"
wobPidFile="/tmp/wob-volume.pid$$"

if [ -z "${wobPipe}" ] || [ -z "${wobConfig}" ]; then
    echo "Usage: $0 PIPE_PATH CONFIG_FILE"
    exit 1
fi

# Cleanup on exit
cleanup() {
    [ -f "$wobPidFile" ] && kill "$(cat "$wobPidFile")" 2>/dev/null || true
    rm -f "$wobPidFile" "$wobPipe"
}
trap cleanup EXIT

# Create FIFO if doesn't exist
[ -p "$wobPipe" ] || mkfifo "$wobPipe"

start_wob() {
    tail -f "$wobPipe" | wob -c "$wobConfig" &
    echo $! > "$wobPidFile"
}

# Kill previous instance (by PID, not killall)
[ -f "$wobPidFile" ] && kill "$(cat "$wobPidFile")" 2>/dev/null

# Start fresh
start_wob

# Watch config file with entr (stdin from heredoc)
entr -r -- sh -c "
    [ -f '$wobPidFile' ] && kill \$(cat '$wobPidFile') 2>/dev/null || true
    tail -f '$wobPipe' | wob -c '$wobConfig' &
    echo \$! > '$wobPidFile'
    wait \$(cat '$wobPidFile')
" <<EOF
$wobConfig
EOF
#+end_src
