* Firefox Emacs keybinds
Remap the defaults.

Firefox doesn't support any sane way of customizing keybinds. So remap
Firefox "locally" (per-application).

- See
  + [[https://support.mozilla.org/en-US/kb/keyboard-shortcuts-perform-firefox-tasks-quickly][Keyboard shortcuts - Perform common Firefox tasks quickly | Firefox Help]]
  + [[https://github.com/k0kubun/xremap/blob/master/example/emacs.yml][k0kubun/xremap/example/emacs.yml]]

#   xremap is acting on the keyboard.

# Credit: https://github.com/mooz/xkeysnail/blob/bf3c93b4fe6efd42893db4e6588e5ef1c4909cfb/example/config.py#L62-L125
keymap:
  - name: Browser Emacs
    application:
      only:
        - firefox
        - vivaldi-stable
        - com.chatterino.chatterino
        - thorium-browser
        - Thorium-browser # X11
        - org.musicbrainz.Picard
        - zen
    exact_match: true # avoid WM conflicts with SUPER+MOD+...
    remap:
      C-b: { with_mark: left }
      C-f: { with_mark: right }
      C-p: { with_mark: up }
      C-n: { with_mark: down }
      M-b: { with_mark: C-left }
      M-f: { with_mark: C-right }
      C-a: { with_mark: home }
      C-e: { with_mark: end }
      M-v: { with_mark: pageup }
      C-v: { with_mark: pagedown }
      M-SHIFT-comma: { with_mark: C-home }
      M-SHIFT-dot: { with_mark: C-end }
      C-m: enter
      C-j: SHIFT-enter
      C-o: [SHIFT-enter, left]
      C-w: [C-x, { set_mark: false }]
      M-w: [C-c, { set_mark: false }]
      C-y: [C-v, { set_mark: false }]
      C-d: [delete, { set_mark: false }]
      M-d: [C-delete, { set_mark: false }]
      C-k: [SHIFT-end, C-x, { set_mark: false }]
      Alt-backspace: [C-backspace, {set_mark: false}]
      C-M-space: [C-SHIFT-right, {set_mark: true}]
      C-slash: [C-z, { set_mark: false }]
      C-SHIFT-ro: C-z
      C-space: { set_mark: true }

      C-s: f3
      C-r: SHIFT-f3
      C-g: [esc, { set_mark: false }]
      M-p: C-pageup
      M-n: C-pagedown
      C-M-k: C-M-SHIFT-delete # close tab
      M-SHIFT-b: M-left
      M-SHIFT-f: M-right
      C-x:
        remap:
          h: [C-home, C-a, { set_mark: true }] # C-x h (select all)
          C-f: C-t # New Tab
          C-w: C-M-SHIFT-insert # Save Page As...
          k: C-f4
          C-c: C-M-SHIFT-q
          u: C-M-SHIFT-slash
          "4":
          - name: Window actions
            remap:
              "0": C-M-SHIFT-end # kill window
          "5":
          - name: Window actions
            remap:
              u: C-M-SHIFT-apostrophe
          t:
            remap:
              u: C-M-SHIFT-slash
          M-b: C-M-SHIFT-b
          M-f: C-M-SHIFT-f
          x:
            remap:
              g: f5
      C-c:
        remap:
          d: C-M-SHIFT-d
          i: C-M-SHIFT-i
          e:
          - name: Extensions
            remap:
              tab: C-M-SHIFT-e
          z:
          - name: Zen Browser features
            application:
              only: zen
            remap:
              c: C-M-SHIFT-c
              s: C-M-SHIFT-s

** Systemd user service
:PROPERTIES:
:CUSTOM_ID: orgid-iztysr
:tangle-dir: (expand-file-name "systemd/user" (xdg-config-home))
:END:

#+begin_src systemd :tangle (tdir "xremap.service")
[Unit]
Description=Keyboard remapper (custom configuration)
After=graphical-session.target
PartOf=graphical-session.target

[Service]
Type=exec
ExecStart=sh -c 'xremap-default --service'
Restart=always
RestartSec=3
SyslogIdentifier=xremap-emacs-browser
StandardOutput=append:%t/xremap.log
StandardError=inherit

[Install]
WantedBy=graphical-session.target
#+end_src


* Launcher script
:PROPERTIES:
:tangle-dir: (xdg-bin-home)
:END:

#+begin_src shell :tangle (tdir "xremap-default") :shebang #!/bin/sh
command -v xremap >/dev/null || {
    notify-send -a "xremap-default script" "ERROR: xremap command not available" "add it to path of the running shell"
    exit 1
}

cmd () {
    xremap ~/.config/xremap/browser-emacs.yaml --device 'SINO WEALTH USB KEYBOARD'
}

notify() {
    notify-send -a "xremap-default script" "xremap killed" "restart it as necessary."
}

trap notify INT QUIT TSTP TERM EXIT

kbname='SINO WEALTH USB KEYBOARD'

if xremap --list-devices 2>/dev/null | grep -q "${kbname}"; then
    xremap ~/.config/xremap/browser-emacs.yaml --device "${kbname}"
else
    notify-send -a "xremap-default" "Device \"${kbname}\" not found, running without --device filter"
    xremap ~/.config/xremap/browser-emacs.yaml
fi
#+end_src
