build with
: emacs -Q --script neron-build.el

: (neron-generate-theme 'dark neron-dark-colors neron-dark-overrides)
: (neron-generate-theme 'light neron-light-colors neron-light-overrides)

it generates =neron-dark-theme.el= and =neron-light-theme.el=.

(defun neron--merge-faces (base overrides)
  (if (not overrides)
      base
    (let ((result (copy-alist base)))
      (dolist (ov overrides)
        (setf (alist-get (car ov) result) (cdr ov)))
      result)))

(defconst neron-gen--substitution-engine
  ;; This block is pasted verbatim into the generated file.
  ;; VARIANT is substituted via string-replace before inserting.
  "
  (letrec
      ((resolve
        (lambda (attrs idx)
          (let (result rest)
            (setq rest attrs)
            (while rest
              (let ((key (pop rest))
                    (val (pop rest)))
                (push key result)
                (cond
                 ;; :inherit must never be color-substituted
                 ((eq key :inherit)
                  (push val result))
                 ;; Palette symbol → actual color at index IDX
                 ((and (symbolp val) (assq val colors))
                  (push (nth idx (assq val colors)) result))
                 ;; Nested plist (only for :box, :underline, etc.)
                 ((and (memq key '(:box :underline :overline :strike-through))
                       (listp val))
                  (push (funcall resolve val idx) result))
                 ;; Anything else (literals, keywords, nil, t) → keep
                 (t
                  (push val result)))))
            (nreverse result)))))

    (apply #'custom-theme-set-faces 'neron-VARIANT
           (mapcar (lambda (face-spec)
                     (let ((face  (car face-spec))
                           (attrs (cdr face-spec)))
                       `(,face
                         ((((min-colors 16777216)) ; graphical / 24-bit
                           ,(funcall resolve attrs 1))
                          (((min-colors 256))      ; 256-color terminal
                           ,(funcall resolve attrs 2))
                          (t                       ; tty fallback
                           ,(funcall resolve attrs 3))))))
                   faces))))\n"
  "Substitution engine block. VARIANT is replaced at generation time.")

(defun neron-gen--options-block (v)
  "Emit defgroup/defcustom/defvar for variant string V."
  (concat
   (format "(defgroup neron-%s nil\n" v)

   "  \"Use different font sizes for some headings and titles.\"\n"

   (format "  :group 'neron-%s)\n\n" v)

   (format "(defcustom neron-%s-height-title-1 1.3\n \"Font size 130%%.\"\n :type 'number\n :group 'neron-%s)\n\n" v v)
   (format "(defcustom neron-%s-height-title-2 1.1\n \"Font size 110%%.\"\n :type 'number\n :group 'neron-%s)\n\n" v v)
   (format "(defcustom neron-%s-height-title-3 1.0\n \"Font size 100%%.\"\n :type 'number\n :group 'neron-%s)\n\n" v v)
   (format "(defcustom neron-%s-height-doc-title 1.44\n \"Font size 144%%.\"\n :type 'number\n :group 'neron-%s)\n\n" v v)

   (format "(defcustom neron-%s-alternate-mode-line-and-minibuffer nil\n" v)
   "  \"Use less bold and pink in the minibuffer.\"\n"
   "  :type 'boolean\n"
   (format "  :group 'neron-%s)\n\n" v)))

(defun neron-generate-theme (variant colors overrides &optional output-dir)
  "Generate a standalone neron-VARIANT-theme.el.

VARIANT is a symbol ('dark or 'light).
COLORS is the palette alist ((name hex-24bit hex-256 tty) ...).
OVERRIDES is a face overrides alist or nil.
OUTPUT-DIR defaults to the directory of this file."
  (let* ((v        (symbol-name variant))
         (V        (capitalize v))
         (out-dir  (or output-dir
                       (file-name-directory
                        (or load-file-name buffer-file-name default-directory))))
         (out-file (expand-file-name (format "neron-%s-theme.el" v) out-dir))
         (merged   (neron--merge-faces (neron--get-base-faces) overrides))
         ;; pp-to-string handles all quoting for us — no manual escaping
         (colors-str (pp-to-string colors))
         (faces-str  (pp-to-string merged))
         ;; Substitute the variant name into the engine block
         (engine   (string-replace "VARIANT" v neron-gen--substitution-engine)))

    (with-temp-file out-file
      ;; File header
      (insert (format ";;; neron-%s-theme.el --- Neron %s Theme  -*- lexical-binding: t; -*-\n\n" v V))
      (insert (format ";; Keywords: theme %s neon neron\n\n" v))
      (insert ";;; Commentary:\n\n")
      (insert ";; Generated file — standalone and editable.\n")
      (insert ";; To regenerate: emacs -Q --script build.el\n\n")
      (insert ";;; Code:\n\n")

      ;; Theme declaration
      (insert (format "(deftheme neron-%s)\n\n" v))

      ;; Options
      (insert ";;;; Configuration options\n\n")
      (insert (neron-gen--options-block v))

      ;; The let block with colors + faces + inline engine
      (insert "(let ((colors '")
      (insert colors-str)
      (insert ")\n\n      (faces '")
      (insert faces-str)
      (insert "))\n")
      (insert engine)

      ;; Autoload for package distribution
      (insert ";;;###autoload\n")
      (insert "(when (and (boundp 'custom-theme-load-path) load-file-name)\n")
      (insert "  (add-to-list 'custom-theme-load-path\n")
      (insert "               (file-name-directory load-file-name)))\n\n")

      ;; Footer
      (insert (format "(provide-theme 'neron-%s)\n" v))
      (insert (format ";;; neron-%s-theme.el ends here\n" v))

      ;; format buffer
      (emacs-lisp-mode)
      (let ((inhibit-message t))        ; optional, keep quiet
        ;; (indent-region (point-min) (point-max))
        (untabify (point-min) (point-max))))

    (message "Generated: %s" out-file)))
